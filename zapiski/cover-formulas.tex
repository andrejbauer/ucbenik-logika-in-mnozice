% Intuitionistic propositional logic
\F{p \lthen (q \lthen p)} % Hilbert's axiom K
\F{(p \lthen (q \lthen r)) \lthen ((p \lthen q) \lthen (p \lthen r))} % Hilbert's axiom S
\F{p \land q \lthen p} % elimination of conjunction, left
\F{p \land q \lthen q} % elimination of conjunction, right
\F{p \lthen (q \lthen p \land q)} % introduction of conjunction
\F{p \lthen p \lor q} % introduction of disjunction, left
\F{q \lthen p \lor q} % introduction of disjunction, right
\F{(p \lthen r) \lthen ((q \lthen r) \lthen (p \lor q \lthen r))} % elimination of disjunction
\F{\bot \lthen p} % ex falso quodlibet
\F{\lnot p \liff (p \lthen \bot)} % negation as implication of falsehood
\F{(p \land q \lthen r) \liff (p \lthen (q \lthen r))} % currying
\F{p \land (q \lor r) \liff (p \land q) \lor (p \land r)} % distributivity of conjunction over disjunction
\F{p \lor (q \land r) \liff (p \lor q) \land (p \lor r)} % distributivity of disjunction over conjunction
\F{\lnot (p \lor q) \liff \lnot p \land \lnot q} % De Morgan's law for disjunction
\F{p \lthen \lnot\lnot p} % introduction of double negation
\F{\lnot\lnot\lnot p \liff \lnot p} % triple negation is negation
\F{\lnot\lnot (p \lor \lnot p)} % double negation of excluded middle

% Classical and intermediate logics
\F{p \lor \lnot p} % law of excluded middle
\F{\lnot\lnot p \lthen p} % elimination of double negation
\F{((p \lthen q) \lthen p) \lthen p} % Peirce's law
\F{\lnot (p \land q) \liff \lnot p \lor \lnot q} % De Morgan's law for conjunction, classical
\F{(p \lthen q) \liff (\lnot q \lthen \lnot p)} % contraposition, classical
\F{(p \lthen q) \lor (q \lthen p)} % Dummett's axiom of Gödel-Dummett logic
\F{\lnot p \lor \lnot\lnot p} % weak excluded middle, axiom of De Morgan logic
\F{(\lnot p \lthen q \lor r) \lthen (\lnot p \lthen q) \lor (\lnot p \lthen r)} % Kreisel-Putnam axiom

% Predicate logic
\F{(\all{x} \phi(x)) \lthen \phi(t)} % universal instantiation
\F{\phi(t) \lthen \some{x} \phi(x)} % existential generalization
\F{\all{x} x = x} % reflexivity of equality
\F{\all{x y} x = y \lthen (\phi(x) \lthen \phi(y))} % Leibniz's law
\F{(\lnot \some{x} \phi(x)) \liff \all{x} \lnot \phi(x)} % De Morgan's law for the existential quantifier
\F{(\lnot \all{x} \phi(x)) \liff \some{x} \lnot \phi(x)} % De Morgan's law for the universal quantifier, classical
\F{(\all{x} \phi(x) \land \psi(x)) \liff (\all{x} \phi(x)) \land (\all{x} \psi(x))} % universal quantifier commutes with conjunction
\F{(\some{x} \phi(x) \lor \psi(x)) \liff (\some{x} \phi(x)) \lor (\some{x} \psi(x))} % existential quantifier commutes with disjunction
\F{(\some{x} \all{y} \rho(x, y)) \lthen \all{y} \some{x} \rho(x, y)} % exchange of quantifiers
\F{\some{x} \phi(x) \lthen \all{y} \phi(y)} % drinker paradox, classical
\F{(\all{x} p \lor \phi(x)) \lthen p \lor \all{x} \phi(x)} % constant domain axiom
\F{(p \lthen \some{x} \phi(x)) \lthen \some{x} (p \lthen \phi(x))} % independence of premise

% Zermelo-Fraenkel set theory
\F{\all{A B} (\all{x} x \in A \liff x \in B) \lthen A = B} % axiom of extensionality
\F{\all{x y z} z \in \set{x, y} \liff z = x \lor z = y} % axiom of pairing
\F{\all{A x} x \in \bigcup A \liff \some{B \in A} x \in B} % axiom of union
\F{\all{x} x \notin \emptyset} % axiom of the empty set
\F{\some{A} \emptyset \in A \land \all{x \in A} x \cup \set{x} \in A} % axiom of infinity
\F{\all{A y} y \in \set{x \in A \mid \phi(x)} \liff y \in A \land \phi(y)} % axiom schema of separation
\F{\all{A S} S \in \pow{A} \liff S \subseteq A} % axiom of power set
\F{\all{A} (\all{x \in A} \exactlyone{y} \phi(x, y)) \lthen \some{B} \all{y} y \in B \liff \some{x \in A} \phi(x, y)} % axiom schema of replacement
\F{\all{A} A \neq \emptyset \lthen \some{x \in A} x \cap A = \emptyset} % axiom of foundation
\F{\all{A} \emptyset \notin A \lthen \some{f : A \to \bigcup A} \all{B \in A} f(B) \in B} % axiom of choice

% Sets
\F{\all{A B} A \subseteq B \liff \all{x} x \in A \lthen x \in B} % definition of subset
\F{\all{A B x} x \in A \cup B \liff x \in A \lor x \in B} % definition of union
\F{\all{A B x} x \in A \cap B \liff x \in A \land x \in B} % definition of intersection
\F{\all{A B x} x \in A \setminus B \liff x \in A \land x \notin B} % definition of difference
\F{\all{A B C} A \cap (B \cup C) = (A \cap B) \cup (A \cap C)} % distributivity of intersection over union
\F{\all{A B C} A \setminus (B \cup C) = (A \setminus B) \cap (A \setminus C)} % De Morgan's law for union
\F{\all{A B C} A \setminus (B \cap C) = (A \setminus B) \cup (A \setminus C)} % De Morgan's law for intersection
\F{\all{A} \emptyset \subseteq A} % the empty set is a subset of every set
\F{\pow{\emptyset} = \set{\emptyset}} % power set of the empty set
\F{\all{x y} (x, y) = \set{\set{x}, \set{x, y}}} % Kuratowski's ordered pair
\F{\all{x y u v} (x, y) = (u, v) \liff x = u \land y = v} % characteristic property of ordered pairs
\F{\all{A B} A \times B = \set{(x, y) \mid x \in A \land y \in B}} % definition of cartesian product
\F{\all{f g h} h \circ (g \circ f) = (h \circ g) \circ f} % associativity of composition
\F{\lnot \some{R} \all{x} x \in R \liff x \notin x} % Russell's paradox
\F{\all{x} x \notin x} % no set is its own element, by foundation
\F{(\all{A} (\all{x \in A} \phi(x)) \lthen \phi(A)) \lthen \all{A} \phi(A)} % epsilon-induction
\F{\all{\alpha} V_\alpha = \bigcup_{\beta < \alpha} \pow{V_\beta}} % cumulative hierarchy
\F{\all{A} \all{f : A \to \pow{A}} \some{S \in \pow{A}} \all{x \in A} f(x) \neq S} % Cantor's theorem, no map onto the power set
\F{\all{A B} |A| \leq |B| \land |B| \leq |A| \lthen |A| = |B|} % Cantor-Schröder-Bernstein theorem
\F{\all{A} |A| < |\pow{A}|} % Cantor's theorem for cardinalities
\F{2^{\aleph_0} = \aleph_1} % continuum hypothesis
\F{\all{\alpha} 2^{\aleph_\alpha} = \aleph_{\alpha + 1}} % generalized continuum hypothesis

% Higher set theory
\F{V = L} % axiom of constructibility
\F{\all{\alpha} L_{\alpha + 1} = \mathrm{Def}(L_\alpha)} % constructible hierarchy, successor stage
\F{\all{\lambda} L_\lambda = \bigcup_{\alpha < \lambda} L_\alpha} % constructible hierarchy, limit stage
\F{\some{\kappa} \some{j : V \to V} \mathrm{crit}(j) = \kappa} % Reinhardt cardinal, refuted by Kunen in ZFC

\F{\all{A \subseteq \NN^{\NN}} (\some{\sigma} \all{\beta} \sigma * \beta \in A) \lor \some{\tau} \all{\alpha} \alpha * \tau \notin A} % axiom of determinacy

% Tarski's compositional truth
\F{\all{s t} \mathrm{Tr}(s \mathbin{\dot{=}} t) \liff \mathrm{val}(s) = \mathrm{val}(t)} % truth of an equation between closed terms
\F{\all{\phi \in \mathrm{Sent}} \mathrm{Tr}(\dot{\lnot} \phi) \liff \lnot \mathrm{Tr}(\phi)} % truth of a negation
\F{\all{\phi \psi \in \mathrm{Sent}} \mathrm{Tr}(\phi \mathbin{\dot{\land}} \psi) \liff \mathrm{Tr}(\phi) \land \mathrm{Tr}(\psi)} % truth of a conjunction
\F{\all{\phi \psi \in \mathrm{Sent}} \mathrm{Tr}(\phi \mathbin{\dot{\lor}} \psi) \liff \mathrm{Tr}(\phi) \lor \mathrm{Tr}(\psi)} % truth of a disjunction
\F{\all{\phi \psi \in \mathrm{Sent}} \mathrm{Tr}(\phi \mathbin{\dot{\Rightarrow}} \psi) \liff (\mathrm{Tr}(\phi) \lthen \mathrm{Tr}(\psi))} % truth of an implication
\F{\all{x} \all{\phi} \mathrm{Tr}(\dot{\forall} x\, \phi) \liff \all{n \in \NN} \mathrm{Tr}(\phi[x \mapsto \underline{n}])} % truth of a universal statement
\F{\all{x} \all{\phi} \mathrm{Tr}(\dot{\exists} x\, \phi) \liff \some{n \in \NN} \mathrm{Tr}(\phi[x \mapsto \underline{n}])} % truth of an existential statement
\F{\mathrm{Tr}(\ulcorner \phi \urcorner) \liff \phi} % Tarski's T-schema
\F{\lnot \some{\tau} \all{\phi} \tau(\ulcorner \phi \urcorner) \liff \phi} % Tarski's undefinability of truth

% Peano arithmetic
\F{\all{n \in \NN} n + 1 \neq 0} % zero is not a successor
\F{\all{m n \in \NN} m + 1 = n + 1 \lthen m = n} % the successor is injective
\F{\all{S \subseteq \NN} 0 \in S \land (\all{n \in S} n + 1 \in S) \lthen S = \NN} % induction
\F{\all{m \in \NN} m + 0 = m} % addition, base case
\F{\all{m n \in \NN} m + (n + 1) = (m + n) + 1} % addition, successor case
\F{\all{m \in \NN} m \cdot 0 = 0} % multiplication, base case
\F{\all{m n \in \NN} m \cdot (n + 1) = m \cdot n + m} % multiplication, successor case
\F{\mathsf{PA} \nvdash \mathsf{Con}(\mathsf{PA})} % Gödel's second incompleteness theorem

% Constructive mathematics and Brouwerian principles

\F{(\all{n \in \NN} \some{m \in \NN} \phi(n, m)) \lthen \some{f \in \NN^{\NN}} \all{n \in \NN} \phi(n, f(n))} % number-number choice

\F{\all{B \subseteq 2^{<\NN}} (\all{\alpha \in 2^{\NN}} \some{n \in \NN} \overline{\alpha}n \in B) \lthen \some{m \in \NN} \all{\alpha \in 2^{\NN}} \some{n \leq m} \overline{\alpha}n \in B} % Brouwer's fan theorem
\F{\all{f : \RR \to \RR} \all{x \in \RR} \all{\varepsilon > 0} \some{\delta > 0} \all{y \in \RR} |x - y| < \delta \lthen |f(x) - f(y)| < \varepsilon} % Brouwer's theorem, every real function is continuous
\F{\all{f \in \NN^{\NN}} \some{e \in \NN} \all{n \in \NN} \some{k \in \NN} T(e, n, k) \land U(k) = f(n)} % Church's thesis, with Kleene's T and U

\F{\some{\alpha \in 2^{\NN}} p \liff \some{n \in \NN} \alpha(n) = 1} % Kripke's schema
\F{\all{x y \in \RR} \lnot\lnot (x = y) \lthen x = y} % stability of equality of reals

\F*{\some{W : \NN \to \mathcal{E}} \all{A \in \mathcal{E}} \some{n \in \NN} W_n = A} % enumeration axiom of synthetic computability

% Computability theory
\F{W_e \defeq \set{n \in \NN \mid \varphi_e(n)\downarrow}} % the e-th computably enumerable set
