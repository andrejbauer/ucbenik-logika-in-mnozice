\chapter{Razredi in družine}
\label{cha:razred-druzine}

\section{Russellov paradoks}

V prejšnji lekciji smo spoznali zapis podmnožice
%
\begin{equation*}
    \set{ x \in A \such \phi(x) },
\end{equation*}
%
ki tvori podmnožico $A$ vseh elementov, ki zadoščajo pogoju $\phi$. Ko je bila
teorija množic še v povojih, se je sama po sebi ponujala ideja, da bi lahko
opisali množice kot ">kakršnokoli"< zbirko stvari. Torej bi lahko pisali
%
\begin{equation*}
    \set{ x \such \phi(x) }
\end{equation*}
%
za množico vseh tistih stvari (objektov, matematičnih entitet), ki zadoščajo
pogoju~$\phi$:
%
\begin{equation*}
    a \in \set{ x \such \phi(x) } \liff \phi(a)
\end{equation*}
%
A je znameniti filozof, logik in matematik Bertrand Russell odkril, 
da je taka, neomejena tvorba zbirk protislovna.
%
Razmislek, ki se po njem imenuje \df{Russellov paradoks}, je v matematiko vnesel pravo ">krizo
temeljev"<, iz katere je v prvi polovici 20.~stoletja izšla logika in
temelji matematike, kot jih poznamo danes.

Russellov paradoks gre takole. Denimo, da bi lahko tvorili množico vseh množic, ki niso element same
sebe:
%
\begin{equation*}
    R \defeq \set{ S \such S \not\in S }.
\end{equation*}
%
Tedaj bi lahko izpeljali protislovje, ker bi hkrati veljalo $R \not\in R$ in $R \in R$:
%
\begin{enumerate}
\item Dokažimo $R \not\in R$.
  %
  Denimo, da bi veljalo $R \in R$. Potem po definiciji $R$ velja $R \not\in R$, kar
  je v protislovju s predpostavko $R \in R$.

\item
  Dokažimo $R \in R$. V prvem koraku smo že dokazali $R \not\in R$, torej po
  definiciji $R$ velja $R \in R$.
\end{enumerate}


\section{Množice in razredi}

V sodobni teoriji množic Russellov paradoks razrešimo tako, da ločimo med dvema
zvrstema zbirk ali skupkov elementov, namreč \df{množicami} in \df{razredi}:
%
\begin{enumerate}
\item \df{urelementi} so matematični objekti, ki niso skupki,
\item \df{množice} so skupki, katerih elementi so urelementi in množice,
\item \df{razredi} so skupki, katerih elementi so urelementi in množice.
\end{enumerate}
%
Na prvi pogled ni razlike med množicami in razredi, a to ni res:
\emph{množica je lahko element množice ali razreda,
razred pa ne more biti element množice ali razreda.}
%
Zapis
%
\begin{equation*}
    x \in Y
  \end{equation*}
%
je veljaven v primeru, da je $x$ urelement ali množica in $Y$ množica ali razred.
Če je $x$ razred, tedaj $x \in Y$ ni veljavna izjava, podobno kot $1/0$ ni veljaven izraz, in sploh ne moremo govoriti o njegovi resničnostni vrednosti.

Sedaj lahko natančneje povemo, kako tvorimo razrede. Razred
%
\begin{equation*}
  \set{ x \such \phi(x) }
\end{equation*}
%
je skupek vseh urelementov in množic, ki zadoščajo pogoju $\phi$:
%
\begin{equation*}
    a \in \set{ x \such \phi(x) } \iff \phi(a).
\end{equation*}
%
Predpišimo tudi \df{pravilo ekstenzionalnosti za razrede}, ki pravi da sta razreda~$C$ in~$D$ enaka, če imata iste elemente:
%
\begin{equation*}
  C = D \liff \all{x} (x \in C \liff x \in D).
\end{equation*}

\begin{zgled}
  \df{Russellov razred} $R \defeq \set{ S \such S \not\in S }$ vsebuje vse množice, ki niso element same sebe. Paradoks smo razrešili, saj je nesmiselno zapisati $R \in R$.
\end{zgled}

\begin{zgled}
  Razred $\set{x \such \bot}$ je prazni razred, saj nima elementov.
\end{zgled}

\begin{zgled}
  Razred $\set{x \such x = 42}$ vsebuje natanko en element, namreč število~$42$.
\end{zgled}

\begin{zgled}
  Naj bo $S$ množica. Razred $\set{x \such x \in S}$ vsebuje natanko elemente množice~$S$.
\end{zgled}

Zadnji zgled pove, da so nekateri razredi pravzaprav množice. Pravimo, da je $C$ \df{nepravi razred}, če obstaja množica $S$, ki ima iste elemente kot~$C$, velja $\all{x} (x \in S \liff x \in C)$. V nasprotnem primeru je $C$ \df{pravi razred}.

\begin{zgled}
  \df{Razred vseh množic}
  %
  \begin{equation*}
    \Set \defeq \set{ S \such \text{$S$ je množica} },
  \end{equation*}
  %
  ki ga označimo tudi z $V$, je pravi razred. Res, če bi bil $V$ množica,
  potem bi lahko tvorili \emph{množico}
  $\set{ S \in V \such S \not\in S}$,
  ki ni nič drugega kot Russellov protislovni razred~$R$.
\end{zgled}

\begin{zgled}
  Razred vseh enojcev $E \defeq \set{ S \such \exactlyone{x} x \in S }$ je pravi razred. Res, če bi bil množica, potem bi bila množica tudi njegova unija $\bigcup E$, ki pa je enaka~$V$.
\end{zgled}

Z razredi lahko delamo tako kot z množicami: tvorimo unije, preseke in produkte
razredov in govorimo o podrazredih. Pri tem uporabljamo enake oznake za
operacije kot pri množicah. Paziti moramo le, da razreda nikoli ne uporabimo kot
element kake množice ali razreda. Na primer, če je $C$ razred, lahko tvorimo
potenčni razred $\pow{C}$, ki vsebuje vse \emph{podmnožice} $C$:
%a
\begin{equation*}
  \pow{C} \defeq \set{ S \such S \in \Set \land S \subseteq C }.
\end{equation*}

\section{Družine množic}

Pogosto imamo opravka z zbirko množic. Če je zbirka končna, lahko množice preprosto
naštejemo in vsako od njih poimenujemo
%
\begin{align*}
    A &= \cdots \\
    B &= \cdots \\
    C &= \cdots
\end{align*}
%
Če je množic neskončno, jih morda lahko oštevilčimo:
%
\begin{align*}
    A_1 &= \cdots \\
    A_2 &= \cdots \\
    A_3 &= \cdots \\
    A_4 &= \cdots \\
        &\vdots
\end{align*}
%
A tu se zadeve še ne končajo, saj lahko v splošnem obravnavamo poljubno zbirko množic.
Takim zbirkam pravimo \df{družine množic}. Družina množic je \df{indeksirana} z elementi neke množice $I$, ki ji pravimo \df{indeksna množica}. Za vsak $i \in I$ imamo množico $A_i$,  kar lahko izrazimo tudi z naslednjo definicijo.

\begin{definicija}
  \df{Družina množic} je preslikava $I \to \Set$. Množici $I$ pravimo \df{indeksna množica} in njenim elementom \df{indeksi}.
\end{definicija}


\begin{zgled}
  Končno zbirko množic lahko indeksiramo s končno množico. Denimo, da imamo
  množice $A$, $B$, $C$, $D$, $E$. Iz njih lahko tvorimo družino $S : I \to \Set$:
  %
  \begin{align*}
  I &= \set{1, 2, 3, 4, 5}, \\
  S_1 &= A,\\
  S_2 &= B,\\
  S_3 &= C,\\
  S_4 &= D,\\
  S_5 &= E.
  \end{align*}
\end{zgled}

\begin{zgled}
  Nihče nas ne sili, da morajo biti indeksi števila. V prejšnjem primeru bi lahko
  uporabili $I = \set{42, 13, \sqrt{2}, \emptyset, \RR}$ in definirali $S : I \to \Set$
  %
  \begin{align*}
  S_{42} &= A,\\
  S_{13} &= B,\\
  S_{\sqrt{2}} &= C,\\
  S_{\emptyset} &= D,\\
  S_{\RR} &= E.
  \end{align*}
\end{zgled}


\begin{zgled}
  Množice v družini se lahko ponavljajo. Skrajni primer je \df{konstantna družina}, v kateri so vse množice med seboj enake.
\end{zgled}

\begin{zgled}
  \df{Prazna družina} je družina množic, ki je indeksirana z $\emptyset$.
\end{zgled}

\begin{zgled}
  Prazno družino moramo ločiti od \df{družine praznih množic}
  %
  \begin{align*}
    I &\to \Set \\
    i &\mapsto \emptyset
  \end{align*}
\end{zgled}

\begin{zgled}
  \df{Neprazna družina} je družina indeksirana z neprazno množico.
  \df{Družina nepraznih množic} je družina, v kateri so vse množice neprazne.
  Torej velja:
  %
  \begin{itemize}
  \item Prazna družina je družina nepraznih množic.
  \item Družina praznih množic je lahko prazna družina (ko je indeksna množica $\emptyset$).
  \item Družina praznih množic je lahko neprazna družina (ko je indeksna množica neprazna).
  \end{itemize}
\end{zgled}


\section{Konstrukcije in operacije z družinami množic}

Operacije $\times$, $+$, $\cap$ in $\cup$ lahko posplošimo tako, da namesto z dvema
množicama delujejo na poljubnem številu množic. V ta namen uporabimo družine
množic.

\subsection{Presek in unija družine}

Presek in unija družine $A : I \to \Set$ sta definirana takole:
%
\begin{align*}
  \bigcup_{i \in I} A_i &\defeq \set{ x \such \some{i \in I} x \in A_i },
  \\
  \bigcap_{i \in I} A_i &\defeq \set{ x \such \all{ i \in I} x \in A_i }.
\end{align*}
%
Pozor! Na desni strani imamo razred! Res se lahko zgodi, da dobimo pravi razred, namreč
kot presek prazne družine:
%
\begin{align*}
  \bigcap_{i \in \emptyset} A_i
  &= \set{ x \such \all{ i \in \emptyset} x \in A_i } \\
  &= \set { x \such \top } \\
  &= V.
\end{align*}
%
Kdaj pa dobimo množico? Presek neprazne družine je vedno množica. Res, če imamo
$k \in I$, potem velja
%
\begin{equation*}
    \bigcap_{i \in I} A_i = \set{ x \in A_k \such \all{i \in I} x \in A_i}.
\end{equation*}
%
Sedaj na desni ne stoji več razred, ampak podmnožica množice~$A_k$.

Kaj pa unija družine množic? Ali je množica? Izkaže se, da za to potrebujemo aksiom:

\begin{aksiom}
  Unija družine množic je množica.
\end{aksiom}


\subsection{Kartezični produkt družine}

\begin{definicija}
  \df{Funkcija izbire} za družino $A : I \to \Set$ je tako prirejanje, ki vsakemu indeksu $i \in I$ priredi natanko en element $f(i) \in A_i$.
 \end{definicija}

 \begin{zgled}
   Primer: funkcija izbire za družino
   %
   \begin{align*}
   A &: \NN \to \Set \\
   A_n &\defeq \set{ x \in \RR \such 0 < x < 2^{-n} }
   \end{align*}
   %
   je na primer $f(n) \defeq 2^{-n - 1}$. To ni edina funkcija izbire za $A$, lahko bi vzeli tudi $f(n) = 2^{-n} / 3$.
 \end{zgled}

 \begin{definicija}
   \df{Kartezični produkt} družine $A : I \to \Set$ je množica vseh funkcij izbire družine~$A$:
   %
   \begin{equation*}
     \prod_{i \in I} A_i \defeq
     \set{f : I \to \textstyle\bigcup_{i \in I} A_i \such \all{i \in I} f(i) \in A_i}.
   \end{equation*}
   %
   Za vsak $j \in I$ imamo \df{$j$-to projekcijo}
   %
   \begin{align*}
    \pr[j] &:  \left(\textstyle\prod_{i \in I} A_i\right) \to A_j, \\
    \pr[j] &:  f \mapsto f(j).
   \end{align*}
 \end{definicija}

Običajni kartezični produkt dveh množic je poseben primer produkta množic, namreč družine
množic, ki je indeksirana z $I = \set{1, 2}$. Natančneje, velja
%
\begin{equation*}
  A \times B \cong \textstyle\prod_{i \in \set{1, 2}} C_i,
\end{equation*}
%
kjer je $C_1 = A$ in $C_2 = B$.

Tudi eksponentna množica je poseben primer produkta množic, saj velja
%
\begin{equation*}
  B^A \cong \textstyle\prod_{a \in A} B
\end{equation*}
%
Na desni imamo produkt konstantne družine množic
\begin{align*}
  A &\to \Set, \\
  a &\mapsto B.
\end{align*}


\subsection{Koprodukt ali vsota množic}

Vsoto množic posplošimo na koprodukt družine.

\begin{definicija}
  \df{Koprodukt} ali \df{vsota družine} $A : I \to \Set$ je množica
  $\sum_{i \in I} A_i$, katere elementi so $\inj[i](a)$ za $i \in I$ in $a \in A_i$.
  Preslikavi $\inj[k] : A_k \to \sum_{i \in I} A_i$ pravimo \df{$k$-ta injekcija}.

  Poleg tega definiramo še \df{projekciji}
  %
  \begin{align*}
    \fst (\inj[i](a)) &= i, \\
    \snd (\inj[i](a)) &= a.
  \end{align*}
  %
  Namesto $\sum$ se piše tudi $\coprod$.
\end{definicija}

Poseben primer koprodukta je vsota $A + B$, saj velja
%
\begin{equation*}
  A + B \cong \textstyle\sum_{k \in \set{1, 2}} C_k
\end{equation*}
%
kjer je
\begin{align*}
  C &: \set{1, 2} \to \Set \\
  C_1 &\defeq A, \\
  C_2 &\defeq B.
\end{align*}
%
Tudi kartezični produkt $A \times B$ je poseben primer koprodukta, saj velja
%
\begin{equation*}
    A \times B \cong \textstyle\sum_{a \in A} B
\end{equation*}
%
Na desni imamo tokrat koprodukt konstantne družine množic
%
\begin{align*}
  A &\to \Set, \\
  a &\mapsto B.
\end{align*}
